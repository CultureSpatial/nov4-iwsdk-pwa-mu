export function trackMetaEvent(eventName: string, eventData?: Record<string, any>) {
  if (typeof window !== 'undefined' && window.fbq) {

#### Conversions API (Server-Side)

\`\`\`typescript
// app/api/meta-conversion/route.ts
import { NextRequest, NextResponse } from 'next/server'
import crypto from 'crypto'

const FACEBOOK_PIXEL_ID = process.env.META_PIXEL_ID!
const FACEBOOK_ACCESS_TOKEN = process.env.META_CONVERSIONS_API_TOKEN!

export async function POST(req: NextRequest) {
  const { event_name, user_data, custom_data } = await req.json()
  
  // Hash PII for privacy compliance
  const hashedUserData = {
    em: user_data.email ? hashSHA256(user_data.email) : undefined,
    ph: user_data.phone ? hashSHA256(user_data.phone) : undefined,
    fn: user_data.first_name ? hashSHA256(user_data.first_name) : undefined,
    ln: user_data.last_name ? hashSHA256(user_data.last_name) : undefined,
  }
  
  const payload = {
    data: [
      {
        event_name,
        event_time: Math.floor(Date.now() / 1000),
        user_data: hashedUserData,
        custom_data,
        event_source_url: req.headers.get('referer') || 'https://emeraldgem.ai',
        action_source: 'website'
      }
    ]
  }
  
  const response = await fetch(
    `https://graph.facebook.com/v18.0/${FACEBOOK_PIXEL_ID}/events?access_token=${FACEBOOK_ACCESS_TOKEN}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    }
  )
  
  return NextResponse.json(await response.json())
}

function hashSHA256(data: string): string {
  return crypto.createHash('sha256').update(data.toLowerCase().trim()).digest('hex')
}
